#include <iostream>

#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
#include <string>
#include <chrono>

using namespace std;

// ============================================================
// Simulated TCP Socket using a thread-safe message queue
// ============================================================

class SimulatedSocket {
private:
    queue<string> messages;
    mutex mtx;
    condition_variable cv;

public:
    // Simulates send()
    void send(const string& message) {
        {
            lock_guard<mutex> lock(mtx);
            messages.push(message);
        }

        // Wake up the receiving thread
        cv.notify_one();
    }

    // Simulates recv()
    string receive() {
        unique_lock<mutex> lock(mtx);

        // Wait until a message is available
        cv.wait(lock, [&]() {
            return !messages.empty();
        });

        string message = messages.front();
        messages.pop();

        return message;
    }
};

// Two communication channels:
// Client -> Server
SimulatedSocket clientToServer;

// Server -> Client
SimulatedSocket serverToClient;


// ============================================================
// SERVER THREAD
// ============================================================

void server() {

    cout << "\n[SERVER] Server thread started.\n";

    for (int i = 1; i <= 3; i++) {

        // Simulate context switching
        this_thread::sleep_for(chrono::milliseconds(300));

        cout << "[SERVER] Waiting for client message...\n";

        // Simulates recv()
        string message = clientToServer.receive();

        cout << "[SERVER] Received: " << message << endl;

        // Process request
        string response;

        if (message == "Hello Server") {
            response = "Hello Client!";
        }
        else if (message == "How are you?") {
            response = "Server is running successfully.";
        }
        else if (message == "Bye") {
            response = "Goodbye Client!";
        }
        else {
            response = "Unknown request.";
        }

        // Simulates processing time
        this_thread::sleep_for(chrono::milliseconds(200));

        cout << "[SERVER] Sending: " << response << endl;

        // Simulates send()
        serverToClient.send(response);
    }

    cout << "[SERVER] Server thread finished.\n";
}


// ============================================================
// CLIENT THREAD
// ============================================================

void client() {

    cout << "[CLIENT] Client thread started.\n";

    string messages[] = {
        "Hello Server",
        "How are you?",
        "Bye"
    };

    for (int i = 0; i < 3; i++) {

        // Simulates context switching
        this_thread::sleep_for(chrono::milliseconds(500));

        cout << "\n[CLIENT] Sending: "
             << messages[i] << endl;

        // Simulates send()
        clientToServer.send(messages[i]);

        cout << "[CLIENT] Waiting for server response...\n";

        // Simulates recv()
        string response = serverToClient.receive();

        cout << "[CLIENT] Received: "
             << response << endl;
    }

    cout << "[CLIENT] Client thread finished.\n";
}


// ============================================================
// MAIN FUNCTION
// ============================================================

int main() {

    cout << "=============================================\n";
    cout << " TCP CLIENT-SERVER SIMULATION\n";
    cout << " Socket Programming + Multithreading\n";
    cout << " Context Switching Demonstration\n";
    cout << "=============================================\n";

    cout << "\n[MAIN] Creating client and server threads...\n";

    // Create two independent threads
    thread serverThread(server);
    thread clientThread(client);

    /*
        join() waits for the threads to finish.

        The operating system scheduler switches between
        clientThread and serverThread.

        This demonstrates context switching:
        
        Client -> Server -> Client -> Server -> ...
    */

    clientThread.join();
    serverThread.join();

    cout << "\n[MAIN] Both threads completed.\n";
    cout << "[MAIN] Program terminated successfully.\n";

    return 0;
}

#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <pthread.h>

#define PORT 8080

void *server(void *arg)
{
    int serverSocket;
    int clientSocket;

    struct sockaddr_in serverAddr;
    struct sockaddr_in clientAddr;

    socklen_t clientSize = sizeof(clientAddr);

    char buffer[1024];

    (void)arg;

    serverSocket = socket(AF_INET, SOCK_STREAM, 0);

    if (serverSocket < 0)
    {
        perror("Server socket creation failed");
        return NULL;
    }

    serverAddr.sin_family = AF_INET;
    serverAddr.sin_addr.s_addr = INADDR_ANY;
    serverAddr.sin_port = htons(PORT);

    if (bind(serverSocket,
             (struct sockaddr *)&serverAddr,
             sizeof(serverAddr)) < 0)
    {
        perror("Bind failed");
        close(serverSocket);
        return NULL;
    }

    if (listen(serverSocket, 1) < 0)
    {
        perror("Listen failed");
        close(serverSocket);
        return NULL;
    }

    printf("[SERVER] Waiting for client...\n");

    clientSocket = accept(
        serverSocket,
        (struct sockaddr *)&clientAddr,
        &clientSize
    );

    if (clientSocket < 0)
    {
        perror("Accept failed");
        close(serverSocket);
        return NULL;
    }

    printf("[SERVER] Client connected!\n");

    memset(buffer, 0, sizeof(buffer));

    recv(clientSocket, buffer, sizeof(buffer) - 1, 0);

    printf("[SERVER] Received: %s\n", buffer);

    strcpy(buffer, "Hello from Server!");

    send(clientSocket, buffer, strlen(buffer), 0);

    printf("[SERVER] Response sent.\n");

    close(clientSocket);
    close(serverSocket);

    return NULL;
}


void *client(void *arg)
{
    int clientSocket;

    struct sockaddr_in serverAddr;

    char buffer[1024];

    (void)arg;

    sleep(1);

    clientSocket = socket(AF_INET, SOCK_STREAM, 0);

    if (clientSocket < 0)
    {
        perror("Client socket creation failed");
        return NULL;
    }

    serverAddr.sin_family = AF_INET;
    serverAddr.sin_port = htons(PORT);

    serverAddr.sin_addr.s_addr = inet_addr("127.0.0.1");

    if (connect(clientSocket,
                (struct sockaddr *)&serverAddr,
                sizeof(serverAddr)) < 0)
    {
        perror("Connection failed");
        close(clientSocket);
        return NULL;
    }

    printf("[CLIENT] Connected to server!\n");

    strcpy(buffer, "Hello from Client!");

    send(clientSocket, buffer, strlen(buffer), 0);

    printf("[CLIENT] Message sent.\n");

    memset(buffer, 0, sizeof(buffer));

    recv(clientSocket, buffer, sizeof(buffer) - 1, 0);

    printf("[CLIENT] Received: %s\n", buffer);

    close(clientSocket);

    return NULL;
}


int main(void)
{
    pthread_t serverThread;
    pthread_t clientThread;

    printf("====================================\n");
    printf(" TCP CLIENT-SERVER PROGRAM\n");
    printf("====================================\n");

    if (pthread_create(&serverThread,
                       NULL,
                       server,
                       NULL) != 0)
    {
        perror("Server thread creation failed");
        return 1;
    }

    if (pthread_create(&clientThread,
                       NULL,
                       client,
                       NULL) != 0)
    {
        perror("Client thread creation failed");
        return 1;
    }

    pthread_join(serverThread, NULL);
    pthread_join(clientThread, NULL);

    printf("\nProgram finished.\n");

    return 0;
}
